```go
// SetProgressUpdateCallback 设置进度更新回调（用于 WebSocket 推送）
func (s *TaskScheduler) SetProgressUpdateCallback(callback func(task *Task)) {
    s.mu.Lock()
    defer s.mu.Unlock()
    s.onProgressUpdate = callback
}

// notifyTaskUpdate 通知任务更新
func (s *TaskScheduler) notifyTaskUpdate(task *Task) {
    s.mu.RLock()
    callback := s.onTaskUpdate
    s.mu.RUnlock()
    if callback != nil {
        callback(task)
    }
    // 通过进度更新回调推送 WebSocket 通知
    if s.onProgressUpdate != nil {
        s.onProgressUpdate(task)
    }
}
```

```typescript
// 连接 WebSocket
function connectWebSocket() {
  if (usePolling.value) {
    startPolling()
    return
  }

  try {
    const socket = new WebSocket(createWebSocketUrl('/api/v1/progress'))

    socket.onopen = () => {
      console.log('WebSocket 连接成功')
      reconnectAttempts.value = 0
    }

    socket.onmessage = (event) => {
      const data = JSON.parse(event.data)
      handleWebSocketMessage(data)
    }

    socket.onclose = () => {
      // 自动重连（带退避）
      if (reconnectAttempts.value < maxReconnectAttempts && !usePolling.value) {
        const delay = Math.min(1000 * Math.pow(2, reconnectAttempts.value), 30000)
        reconnectAttempts.value++
        setTimeout(() => {
          if (!usePolling.value) {
            connectWebSocket()
          }
        }, delay)
      }
    }

    ws.value = socket
  } catch (e) {
    // 降级到轮询
    usePolling.value = true
    startPolling()
  }
}
```

### 5. 加权平滑算法（避免进度跳动）

**实现原理**:
- 维护最近 N 次进度历史
- 为每次进度分配不同权重（最近的权重更大）
- 计算加权平均值作为平滑后的进度

**代码片段**:
```typescript
// 加权平滑算法 - 避免进度跳动
function smoothProgress(taskId: number, newProgress: number): number {
  const defaultWeights = [0.4, 0.3, 0.2, 0.1] // 权重总和为 1
  
  if (!cache.has(taskId)) {
    cache.set(taskId, {
      smoothed: newProgress,
      history: [newProgress],
      weights: defaultWeights
    })
    return newProgress
  }
  
  const data = cache.get(taskId)!
  
  // 更新历史
  data.history.push(newProgress)
  if (data.history.length > historySize) {
    data.history.shift()
  }
  
  // 计算加权平均
  let weightedSum = 0
  let weightTotal = 0
  
  for (let i = 0; i < data.history.length; i++) {
    const weightIndex = Math.max(0, data.weights.length - (data.history.length - i))
    const weight = data.weights[weightIndex] || 0.1
    weightedSum += data.history[i] * weight
    weightTotal += weight
  }
  
  data.smoothed = weightedSum / weightTotal
  cache.set(taskId, data)
  
  return Math.round(data.smoothed * 100) / 100
}
```

### 6. 离线缓存补发机制

**实现方式**:
- 使用 `lastProgressCache` 缓存每个任务的平滑进度数据
- 组件卸载时不清除缓存，保持进度连续性
- 重连后自动从缓存恢复进度

**代码片段**:
```typescript
// 离线缓存
const lastProgressCache = ref<Map<number, SmoothedProgress>>(new Map())

// 获取平滑后的进度
function getSmoothedProgress(taskId: number, progress: number): number {
  const smoothed = smoothProgress(taskId, progress)
  return smoothed
}
```

```typescript
function startPolling() {
  console.log('开始轮询模式')
  if (pollTimer) {
    clearInterval(pollTimer)
  }
  pollTimer = setInterval(() => {
    fetchTasks()
  }, pollInterval.value)
}

function stopPolling() {
  if (pollTimer) {
    clearInterval(pollTimer)
    pollTimer = null
  }
}
```
